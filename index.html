/** * STAT Course Feedback System * Automatically processes form responses and generates dashboard data * NO SIMULATED STATISTICS - only real data from form submissions * Last updated: 9/7/2026 */ // ============================================================ // 1. CONFIGURATION // ============================================================ const CONFIG = { RESPONSE_SHEET_NAME: 'Form Responses', DASHBOARD_SHEET_NAME: 'Dashboard', PROCESSED_SHEET_NAME: 'Processed Responses', CHART_DATA_SHEET_NAME: 'Chart Data', // Column mappings (based on your Google Form) COLUMNS: { TIMESTAMP: 0, STUDENT_ID: 1, COURSES: 2, EXPECTATION: 3, SURPRISE: 4, SUGGESTION: 5, CLARITY: 6, Q353_COUNTING: 7, Q353_PROB: 8, Q353_GITHUB: 9, Q360_GITHUB_CONF: 10, Q360_SHEETS_CONF: 11, Q360_AI_CONF: 12, Q415_METRICS: 13, Q415_LOGISTIC: 14, Q415_PACE: 15, Q415_F1: 16 } }; // ============================================================ // 2. TRIGGER: Run automatically on form submission // ============================================================ function installTrigger() { const ss = SpreadsheetApp.getActiveSpreadsheet(); // Remove existing triggers const triggers = ScriptApp.getProjectTriggers(); triggers.forEach(trigger => { if (trigger.getHandlerFunction() === 'onFormSubmit') { ScriptApp.deleteTrigger(trigger); } }); // Create new trigger ScriptApp.newTrigger('onFormSubmit') .forSpreadsheet(ss) .onFormSubmit() .create(); Logger.log('✅ Trigger installed successfully!'); } function onFormSubmit(e) { try { const values = e.values; const ss = SpreadsheetApp.getActiveSpreadsheet(); // Process the new response processResponse(values, ss); // Rebuild the dashboard with real data only refreshDashboard(ss); Logger.log('✅ Form submission processed: Row ' + e.range.getRow()); } catch (error) { Logger.log('❌ Error: ' + error.message); } } // ============================================================ // 3. PROCESS INDIVIDUAL RESPONSE // ============================================================ function processResponse(values, ss) { // Parse courses from checkbox values const coursesRaw = values[CONFIG.COLUMNS.COURSES] || ''; const courses = coursesRaw.split(',').map(c => c.trim()).filter(c => c); // Build structured response object const response = { timestamp: values[CONFIG.COLUMNS.TIMESTAMP], studentId: values[CONFIG.COLUMNS.STUDENT_ID] || 'Anonymous', courses: courses, expectation: values[CONFIG.COLUMNS.EXPECTATION] || '', surprise: values[CONFIG.COLUMNS.SURPRISE] || '', suggestion: values[CONFIG.COLUMNS.SUGGESTION] || '', clarity: values[CONFIG.COLUMNS.CLARITY] || '', // STAT 353 q353_counting: parseFloat(values[CONFIG.COLUMNS.Q353_COUNTING]) || null, q353_prob: parseFloat(values[CONFIG.COLUMNS.Q353_PROB]) || null, q353_github: values[CONFIG.COLUMNS.Q353_GITHUB] || '', // STAT 360 q360_github_conf: parseFloat(values[CONFIG.COLUMNS.Q360_GITHUB_CONF]) || null, q360_sheets_conf: parseFloat(values[CONFIG.COLUMNS.Q360_SHEETS_CONF]) || null, q360_ai_conf: parseFloat(values[CONFIG.COLUMNS.Q360_AI_CONF]) || null, // STAT 415 q415_metrics: values[CONFIG.COLUMNS.Q415_METRICS] || '', q415_logistic: values[CONFIG.COLUMNS.Q415_LOGISTIC] || '', q415_pace: values[CONFIG.COLUMNS.Q415_PACE] || '', q415_f1: values[CONFIG.COLUMNS.Q415_F1] || '' }; // Store in processed sheet storeProcessedResponse(response, ss); return response; } function storeProcessedResponse(response, ss) { let sheet = ss.getSheetByName(CONFIG.PROCESSED_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(CONFIG.PROCESSED_SHEET_NAME); const headers = [ 'Timestamp', 'Student ID', 'Courses', 'Expectation', 'Surprise', 'Suggestion', 'Clarity', '353 Counting', '353 Prob', '353 GitHub', '360 GitHub', '360 Sheets', '360 AI', '415 Metrics', '415 Logistic', '415 Pace', '415 F1' ]; sheet.getRange(1, 1, 1, headers.length).setValues([headers]); sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold'); } const row = [ response.timestamp, response.studentId, response.courses.join(', '), response.expectation, response.surprise, response.suggestion, response.clarity, response.q353_counting, response.q353_prob, response.q353_github, response.q360_github_conf, response.q360_sheets_conf, response.q360_ai_conf, response.q415_metrics, response.q415_logistic, response.q415_pace, response.q415_f1 ]; sheet.appendRow(row); } // ============================================================ // 4. GET ALL REAL RESPONSES // ============================================================ function getAllResponses(ss) { const sheet = ss.getSheetByName(CONFIG.PROCESSED_SHEET_NAME); if (!sheet) return []; const data = sheet.getDataRange().getValues(); if (data.length < 2) return []; const responses = []; for (let i = 1; i < data.length; i++) { const row = data[i]; // Skip completely empty rows if (!row[0] && !row[1] && !row[2]) continue; responses.push({ timestamp: row[0], studentId: row[1], courses: row[2] ? row[2].split(',').map(c => c.trim()) : [], expectation: row[3], surprise: row[4], suggestion: row[5], clarity: row[6], q353_counting: row[7], q353_prob: row[8], q353_github: row[9], q360_github_conf: row[10], q360_sheets_conf: row[11], q360_ai_conf: row[12], q415_metrics: row[13], q415_logistic: row[14], q415_pace: row[15], q415_f1: row[16] }); } return responses; } // ============================================================ // 5. CALCULATE SUMMARY STATISTICS (REAL DATA ONLY) // ============================================================ function calculateSummary(responses) { if (responses.length === 0) { return { total: 0, byCourse: {}, expectationCounts: { exceeded: 0, met: 0, partially: 0, notMet: 0, unsure: 0 }, clarityCounts: { very: 0, mostly: 0, somewhat: 0, veryUnclear: 0 }, q353_counting_avg: 0, q353_prob_avg: 0, q353_github_counts: { 'Smooth / No issues': 0, 'Slightly Difficult': 0, 'Very Difficult / Glitchy': 0, 'I haven\'t tried it yet': 0 }, q360_github_avg: 0, q360_sheets_avg: 0, q360_ai_avg: 0, q415_metrics_counts: { 'understand clearly': 0, 'sort of get it': 0, 'completely lost': 0 }, q415_logistic_counts: { 'Very comfortable': 0, 'Somewhat comfortable': 0, 'just memorizing': 0 }, q415_pace_counts: { 'Too Fast': 0, 'Just Right': 0, 'Too Slow': 0 }, q415_f1_counts: { 'understand clearly': 0, 'know formula but not sure when': 0, 'confused': 0, 'dont remember': 0 }, techPulse: 0, paceFast: 0, paceTotal: 0, expectMetCount: 0, surpriseList: [], suggestionList: [] }; } const summary = { total: responses.length, byCourse: {}, expectationCounts: { exceeded: 0, met: 0, partially: 0, notMet: 0, unsure: 0 }, clarityCounts: { very: 0, mostly: 0, somewhat: 0, veryUnclear: 0 }, q353_counting_avg: 0, q353_prob_avg: 0, q353_github_counts: { 'Smooth / No issues': 0, 'Slightly Difficult': 0, 'Very Difficult / Glitchy': 0, 'I haven\'t tried it yet': 0 }, q360_github_avg: 0, q360_sheets_avg: 0, q360_ai_avg: 0, q415_metrics_counts: { 'understand clearly': 0, 'sort of get it': 0, 'completely lost': 0 }, q415_logistic_counts: { 'Very comfortable': 0, 'Somewhat comfortable': 0, 'just memorizing': 0 }, q415_pace_counts: { 'Too Fast': 0, 'Just Right': 0, 'Too Slow': 0 }, q415_f1_counts: { 'understand clearly': 0, 'know formula but not sure when': 0, 'confused': 0, 'dont remember': 0 }, techPulse: 0, paceFast: 0, paceTotal: 0, expectMetCount: 0, surpriseList: [], suggestionList: [] }; const countingVals = []; const probVals = []; const githubConfVals = []; const sheetsConfVals = []; const aiConfVals = []; responses.forEach(r => { // Course breakdown r.courses.forEach(c => { summary.byCourse[c] = (summary.byCourse[c] || 0) + 1; }); // Expectations if (r.expectation && summary.expectationCounts.hasOwnProperty(r.expectation)) { summary.expectationCounts[r.expectation]++; if (r.expectation === 'met' || r.expectation === 'exceeded') { summary.expectMetCount++; } } // Clarity if (r.clarity && summary.clarityCounts.hasOwnProperty(r.clarity)) { summary.clarityCounts[r.clarity]++; } // STAT 353 if (r.q353_counting !== null && r.q353_counting !== undefined && !isNaN(r.q353_counting)) { countingVals.push(r.q353_counting); } if (r.q353_prob !== null && r.q353_prob !== undefined && !isNaN(r.q353_prob)) { probVals.push(r.q353_prob); } if (r.q353_github && summary.q353_github_counts.hasOwnProperty(r.q353_github)) { summary.q353_github_counts[r.q353_github]++; } // STAT 360 if (r.q360_github_conf !== null && r.q360_github_conf !== undefined && !isNaN(r.q360_github_conf)) { githubConfVals.push(r.q360_github_conf); } if (r.q360_sheets_conf !== null && r.q360_sheets_conf !== undefined && !isNaN(r.q360_sheets_conf)) { sheetsConfVals.push(r.q360_sheets_conf); } if (r.q360_ai_conf !== null && r.q360_ai_conf !== undefined && !isNaN(r.q360_ai_conf)) { aiConfVals.push(r.q360_ai_conf); } // STAT 415 if (r.q415_metrics && summary.q415_metrics_counts.hasOwnProperty(r.q415_metrics)) { summary.q415_metrics_counts[r.q415_metrics]++; } if (r.q415_logistic && summary.q415_logistic_counts.hasOwnProperty(r.q415_logistic)) { summary.q415_logistic_counts[r.q415_logistic]++; } if (r.q415_pace && summary.q415_pace_counts.hasOwnProperty(r.q415_pace)) { summary.q415_pace_counts[r.q415_pace]++; if (r.q415_pace === 'Too Fast') summary.paceFast++; summary.paceTotal++; } if (r.q415_f1 && summary.q415_f1_counts.hasOwnProperty(r.q415_f1)) { summary.q415_f1_counts[r.q415_f1]++; } // Collect open-ended responses if (r.surprise) summary.surpriseList.push(r.surprise); if (r.suggestion) summary.suggestionList.push(r.suggestion); }); // Calculate averages summary.q353_counting_avg = countingVals.length ? countingVals.reduce((a, b) => a + b, 0) / countingVals.length : 0; summary.q353_prob_avg = probVals.length ? probVals.reduce((a, b) => a + b, 0) / probVals.length : 0; summary.q360_github_avg = githubConfVals.length ? githubConfVals.reduce((a, b) => a + b, 0) / githubConfVals.length : 0; summary.q360_sheets_avg = sheetsConfVals.length ? sheetsConfVals.reduce((a, b) => a + b, 0) / sheetsConfVals.length : 0; summary.q360_ai_avg = aiConfVals.length ? aiConfVals.reduce((a, b) => a + b, 0) / aiConfVals.length : 0; // Tech pulse (combined average of all tech scores) const allTechVals = [...githubConfVals, ...sheetsConfVals, ...aiConfVals]; summary.techPulse = allTechVals.length ? allTechVals.reduce((a, b) => a + b, 0) / allTechVals.length : 0; return summary; } // ============================================================ // 6. BUILD DASHBOARD // ============================================================ function refreshDashboard(ss) { if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet(); const responses = getAllResponses(ss); if (responses.length === 0) { updateEmptyDashboard(ss); return; } const summary = calculateSummary(responses); updateDashboardSheet(summary, ss); updateChartData(summary, ss); Logger.log('✅ Dashboard updated with ' + responses.length + ' real responses.'); } function updateDashboardSheet(summary, ss) { let sheet = ss.getSheetByName(CONFIG.DASHBOARD_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(CONFIG.DASHBOARD_SHEET_NAME); } else { sheet.clear(); } const now = new Date(); const timezone = Session.getScriptTimeZone(); // ===== HEADER ===== sheet.getRange(1, 1).setValue('📊 STAT Course Feedback Dashboard'); sheet.getRange(2, 1).setValue('Last updated: ' + Utilities.formatDate(now, timezone, 'yyyy-MM-dd HH:mm:ss')); sheet.getRange(1, 1, 2, 1).merge(); sheet.getRange(1, 1, 2, 1).setFontSize(16).setFontWeight('bold'); let row = 4; // ===== SUMMARY STATS ===== sheet.getRange(row, 1).setValue('📈 Summary Statistics'); sheet.getRange(row, 1).setFontWeight('bold').setFontSize(14); row += 2; sheet.getRange(row, 1).setValue('Total Responses:'); sheet.getRange(row, 2).setValue(summary.total); row++; sheet.getRange(row, 1).setValue('Tech Pulse (avg 1-5):'); sheet.getRange(row, 2).setValue(summary.techPulse.toFixed(2)); row++; sheet.getRange(row, 1).setValue('Expectations Met (%):'); const expectPct = summary.total ? (summary.expectMetCount / summary.total * 100).toFixed(1) : 0; sheet.getRange(row, 2).setValue(expectPct + '%'); row++; sheet.getRange(row, 1).setValue('Pace "Too Fast" (415):'); sheet.getRange(row, 2).setValue(summary.paceTotal ? `${summary.paceFast}/${summary.paceTotal}` : 'N/A'); row += 2; // ===== COURSE BREAKDOWN ===== sheet.getRange(row, 1).setValue('📚 Responses by Course'); sheet.getRange(row, 1).setFontWeight('bold').setFontSize(14); row++; const courseHeaders = ['Course', 'Count']; sheet.getRange(row, 1, 1, 2).setValues([courseHeaders]); sheet.getRange(row, 1, 1, 2).setFontWeight('bold'); row++; const courseData = Object.entries(summary.byCourse).sort((a, b) => b[1] - a[1]); if (courseData.length) { sheet.getRange(row, 1, courseData.length, 2).setValues(courseData); row += courseData.length + 2; } // ===== EXPECTATION CHANGE ===== sheet.getRange(row, 1).setValue('🔍 Expectation Change'); sheet.getRange(row, 1).setFontWeight('bold').setFontSize(14); row++; const expHeaders = ['Category', 'Count']; sheet.getRange(row, 1, 1, 2).setValues([expHeaders]); sheet.getRange(row, 1, 1, 2).setFontWeight('bold'); row++; const expData = Object.entries(summary.expectationCounts).filter(([k, v]) => v > 0); if (expData.length) { sheet.getRange(row, 1, expData.length, 2).setValues(expData); row += expData.length + 2; } // ===== CLARITY ===== sheet.getRange(row, 1).setValue('📋 Clarity of Expectations'); sheet.getRange(row, 1).setFontWeight('bold').setFontSize(14); row++; const clarityHeaders = ['Category', 'Count']; sheet.getRange(row, 1, 1, 2).setValues([clarityHeaders]); sheet.getRange(row, 1, 1, 2).setFontWeight('bold'); row++; const clarityData = Object.entries(summary.clarityCounts).filter(([k, v]) => v > 0); if (clarityData.length) { sheet.getRange(row, 1, clarityData.length, 2).setValues(clarityData); row += clarityData.length + 2; } // ===== STAT 353 ===== sheet.getRange(row, 1).setValue('📌 STAT 353'); sheet.getRange(row, 1).setFontWeight('bold').setFontSize(14); row++; sheet.getRange(row, 1).setValue('Average Comfort: Counting Principle (1-5)'); sheet.getRange(row, 2).setValue(summary.q353_counting_avg.toFixed(2)); row++; sheet.getRange(row, 1).setValue('Average Comfort: Basic Probability (1-5)'); sheet.getRange(row, 2).setValue(summary.q353_prob_avg.toFixed(2)); row++; sheet.getRange(row, 1).setValue('GitHub Quiz Experience:'); row++; const ghData = Object.entries(summary.q353_github_counts).filter(([k, v]) => v > 0); if (ghData.length) { sheet.getRange(row, 1, ghData.length, 2).setValues(ghData); row += ghData.length + 2; } // ===== STAT 360 ===== sheet.getRange(row, 1).setValue('🔧 STAT 360 - Tool Confidence (1-5)'); sheet.getRange(row, 1).setFontWeight('bold').setFontSize(14); row++; sheet.getRange(row, 1).setValue('GitHub Confidence:'); sheet.getRange(row, 2).setValue(summary.q360_github_avg.toFixed(2)); row++; sheet.getRange(row, 1).setValue('Sheets Confidence:'); sheet.getRange(row, 2).setValue(summary.q360_sheets_avg.toFixed(2)); row++; sheet.getRange(row, 1).setValue('AI Confidence:'); sheet.getRange(row, 2).setValue(summary.q360_ai_avg.toFixed(2)); row += 2; // ===== STAT 415 ===== sheet.getRange(row, 1).setValue('📉 STAT 415'); sheet.getRange(row, 1).setFontWeight('bold').setFontSize(14); row++; sheet.getRange(row, 1).setValue('Performance Metrics Grasp:'); row++; const metricsData = Object.entries(summary.q415_metrics_counts).filter(([k, v]) => v > 0); if (metricsData.length) { sheet.getRange(row, 1, metricsData.length, 2).setValues(metricsData); row += metricsData.length + 1; } sheet.getRange(row, 1).setValue('Logistic Regression Comfort:'); row++; const logData = Object.entries(summary.q415_logistic_counts).filter(([k, v]) => v > 0); if (logData.length) { sheet.getRange(row, 1, logData.length, 2).setValues(logData); row += logData.length + 1; } sheet.getRange(row, 1).setValue('Pace:'); row++; const paceData = Object.entries(summary.q415_pace_counts).filter(([k, v]) => v > 0); if (paceData.length) { sheet.getRange(row, 1, paceData.length, 2).setValues(paceData); row += paceData.length + 1; } sheet.getRange(row, 1).setValue('F1 Score Understanding:'); row++; const f1Data = Object.entries(summary.q415_f1_counts).filter(([k, v]) => v > 0); if (f1Data.length) { sheet.getRange(row, 1, f1Data.length, 2).setValues(f1Data); } // Auto-resize and format sheet.autoResizeColumns(1, 2); sheet.getRange(1, 1, row + 5, 2).setBorder(true, true, true, true, false, false); } function updateEmptyDashboard(ss) { let sheet = ss.getSheetByName(CONFIG.DASHBOARD_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(CONFIG.DASHBOARD_SHEET_NAME); } else { sheet.clear(); } const now = new Date(); sheet.getRange(1, 1).setValue('📊 STAT Course Feedback Dashboard'); sheet.getRange(2, 1).setValue('Last updated: ' + Utilities.formatDate(now, Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss')); sheet.getRange(1, 1, 2, 1).merge(); sheet.getRange(1, 1, 2, 1).setFontSize(16).setFontWeight('bold'); sheet.getRange(4, 1).setValue('⚠️ No responses received yet.'); sheet.getRange(4, 1).setFontSize(14).setFontColor('#666666'); sheet.getRange(5, 1).setValue('Once students submit the form, the dashboard will populate automatically.'); sheet.getRange(5, 1).setFontSize(12).setFontColor('#888888'); } // ============================================================ // 7. UPDATE CHART DATA // ============================================================ function updateChartData(summary, ss) { let sheet = ss.getSheetByName(CONFIG.CHART_DATA_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(CONFIG.CHART_DATA_SHEET_NAME); } else { sheet.clear(); } let row = 1; // Expectation Change sheet.getRange(row, 1).setValue('Expectation Change'); row++; const expData = Object.entries(summary.expectationCounts).filter(([k, v]) => v > 0); if (expData.length) { sheet.getRange(row, 1, expData.length, 2).setValues(expData); row += expData.length + 2; } // GitHub Experience sheet.getRange(row, 1).setValue('STAT 353 - GitHub Experience'); row++; const ghData = Object.entries(summary.q353_github_counts).filter(([k, v]) => v > 0); if (ghData.length) { sheet.getRange(row, 1, ghData.length, 2).setValues(ghData); row += ghData.length + 2; } // F1 Score sheet.getRange(row, 1).setValue('STAT 415 - F1 Score Understanding'); row++; const f1Data = Object.entries(summary.q415_f1_counts).filter(([k, v]) => v > 0); if (f1Data.length) { sheet.getRange(row, 1, f1Data.length, 2).setValues(f1Data); row += f1Data.length + 2; } // Clarity sheet.getRange(row, 1).setValue('Clarity of Expectations'); row++; const clarityData = Object.entries(summary.clarityCounts).filter(([k, v]) => v > 0); if (clarityData.length) { sheet.getRange(row, 1, clarityData.length, 2).setValues(clarityData); } sheet.autoResizeColumns(1, 2); } // ============================================================ // 8. WEB APP ENDPOINTS (for HTML dashboard) // ============================================================ function doGet(e) { const action = e?.parameter?.action || 'dashboard'; const course = e?.parameter?.course || 'ALL'; if (action === 'dashboard') { return getDashboardJSON(course); } return ContentService.createTextOutput(JSON.stringify({ success: false, error: 'Invalid action' })).setMimeType(ContentService.MimeType.JSON); } function getDashboardJSON(courseFilter) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); let responses = getAllResponses(ss); // Filter by course if specified if (courseFilter !== 'ALL') { responses = responses.filter(r => r.courses.includes(courseFilter)); } if (responses.length === 0) { return ContentService.createTextOutput(JSON.stringify({ success: true, total: 0, message: 'No responses for this filter', stats: null, charts: null })).setMimeType(ContentService.MimeType.JSON); } const summary = calculateSummary(responses); return ContentService.createTextOutput(JSON.stringify({ success: true, total: responses.length, filter: courseFilter, stats: { techPulse: summary.techPulse, expectMetCount: summary.expectMetCount, paceFast: summary.paceFast, paceTotal: summary.paceTotal, q353_counting_avg: summary.q353_counting_avg, q353_prob_avg: summary.q353_prob_avg, q360_github_avg: summary.q360_github_avg, q360_sheets_avg: summary.q360_sheets_avg, q360_ai_avg: summary.q360_ai_avg }, charts: { expectation: summary.expectationCounts, clarity: summary.clarityCounts, github: summary.q353_github_counts, f1: summary.q415_f1_counts, metrics: summary.q415_metrics_counts, logistic: summary.q415_logistic_counts, pace: summary.q415_pace_counts, byCourse: summary.byCourse } })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.message })).setMimeType(ContentService.MimeType.JSON); } } // ============================================================ // 9. ADMIN FUNCTIONS // ============================================================ function setupSystem() { const ss = SpreadsheetApp.getActiveSpreadsheet(); // Create required sheets let sheets = [ CONFIG.PROCESSED_SHEET_NAME, CONFIG.DASHBOARD_SHEET_NAME, CONFIG.CHART_DATA_SHEET_NAME ]; sheets.forEach(name => { if (!ss.getSheetByName(name)) { ss.insertSheet(name); } }); // Install trigger installTrigger(); // Build initial dashboard refreshDashboard(ss); Logger.log('✅ System setup complete!'); Logger.log('📊 Sheets created: ' + sheets.join(', ')); Logger.log('🔔 Trigger installed for automatic updates'); } function clearAllData() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(CONFIG.PROCESSED_SHEET_NAME); if (sheet) { const lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet.deleteRows(2, lastRow - 1); } Logger.log('✅ All data cleared. Headers preserved.'); } refreshDashboard(ss); } function getResponseCount() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const responses = getAllResponses(ss); return responses.length; } function exportCSV() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(CONFIG.PROCESSED_SHEET_NAME); if (!sheet) { Logger.log('❌ No processed responses found.'); return; } const data = sheet.getDataRange().getValues(); const csv = data.map(row => row.join(',')).join('\n'); const date = new Date().toISOString().slice(0, 10); const csvSheet = ss.insertSheet('CSV Export ' + date); csvSheet.getRange(1, 1).setValue(csv); Logger.log('✅ CSV exported to sheet: CSV Export ' + date); }
